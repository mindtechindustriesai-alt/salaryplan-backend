"""Quantum verification routes"""

from fastapi import APIRouter
from datetime import datetime

from app.core.constants import (
    CHSH_SCORE, QUANTUM_CORRELATION, CLASSICAL_LIMIT,
    VIOLATION_PERCENTAGE, IBM_JOB_ID, PATENT_NUMBER,
    PATENT_FILING_DATE
)
from app.models.quantum import QuantumStatusResponse

router = APIRouter(prefix="/quantum", tags=["Quantum"])


@router.get("/status", response_model=QuantumStatusResponse)
async def quantum_status():
    """
    Get quantum verification status
    IBM-verified CHSH Bell test results
    """
    return QuantumStatusResponse(
        quantum_verified=True,
        chsh_score=CHSH_SCORE,
        correlation=QUANTUM_CORRELATION,
        violation_percentage=VIOLATION_PERCENTAGE,
        ibm_job_id=IBM_JOB_ID,
        patent_number=PATENT_NUMBER,
        patent_filing_date=PATENT_FILING_DATE,
        last_verified=datetime.now()
    )


@router.get("/patent")
async def patent_info():
    """Get patent information"""
    return {
        "patent_number": PATENT_NUMBER,
        "filing_date": PATENT_FILING_DATE,
        "status": "provisional",
        "attorney": "ENSafrica - Dr Bernard Dippenaar",
        "title": "Adaptive Learning Ecosystem with Integrated Safety Monitoring and Optional Quantum-Secure Verification"
    }
